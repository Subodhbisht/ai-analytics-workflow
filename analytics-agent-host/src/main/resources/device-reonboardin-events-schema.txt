
TABLE public.ff37_device_reonboard_events (
	id bigserial NOT NULL,
	device_id varchar(64) NOT NULL,
	status varchar(50) NOT NULL,
	event_time timestamp NOT NULL,
	reason varchar(200) NOT NULL,
	bec_name varchar(20) NOT NULL,
	model varchar(50) NOT NULL,
	inserted_at timestamp NOT NULL,
	CONSTRAINT ff37_device_reonboard_events_pkey PRIMARY KEY (id),
	CONSTRAINT uniq_ff37_device_reonboard_event UNIQUE (device_id, status, event_time, reason, bec_name)
);
INDEX idx_ff37_reonboard_bec_model_time ON public.ff37_device_reonboard_events USING btree (bec_name, model, event_time);
INDEX idx_ff37_reonboard_event_time ON public.ff37_device_reonboard_events USING btree (event_time);
INDEX idx_ff37_reonboard_reason ON public.ff37_device_reonboard_events USING btree (reason);
